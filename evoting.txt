    function playTouchSound(type = 'tap') {
      try {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        const ctx = new AudioCtx();
        const now = ctx.currentTime;

        if (type === 'tap') {
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(520, now);
          osc.frequency.exponentialRampToValueAtTime(680, now + 0.08);
          gain.gain.setValueAtTime(0.15, now);
          gain.gain.exponentialRampToValueAtTime(0.01, now + 0.08);
          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(now);
          osc.stop(now + 0.09);
        } else if (type === 'success') {
          // Pleasant two-tone chime for vote recorded
          const notes = [523.25, 659.25, 783.99, 1046.50]; // C5, E5, G5, C6
          notes.forEach((freq, idx) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(freq, now + idx * 0.09);
            gain.gain.setValueAtTime(0.22, now + idx * 0.09);
            gain.gain.exponentialRampToValueAtTime(0.008, now + idx * 0.09 + 0.35);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(now + idx * 0.09);
            osc.stop(now + idx * 0.09 + 0.36);
          });
        }
      } catch (e) {
        // Ignore audio errors if browser blocks autoplay
      }
    }

    // Save Vote & Show Auto-Reset Thank You Screen
    function recordVoteNow() {
      if (!selectedCandidate) return;

      // Increment Vote
      selectedCandidate.votes += 1;
      kioskConfig.lastVoteTime = new Date().toLocaleTimeString('id-ID', {
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit'
      }) + ' WIB';

      saveCandidates();
      saveConfig();

      // Hide Confirm Modal if open
      document.getElementById('confirmModal').classList.add('hidden');

      // Play Chime & Confetti
      playTouchSound('success');
      try {
        confetti({
          particleCount: 90,
          spread: 80,
          origin: { y: 0.6 }
        });
      } catch (e) {}

      // Show Success Modal with Countdown
      document.getElementById('successVotedName').textContent = 
        `Paslon ${selectedCandidate.number} (${selectedCandidate.chairName} & ${selectedCandidate.viceName})`;
      
      let secondsLeft = parseInt(kioskConfig.resetDelay, 10) || 3;
      const timerEl = document.getElementById('countdownTimer');
      timerEl.textContent = secondsLeft;

      document.getElementById('successModal').classList.remove('hidden');
      lucide.createIcons();

      if (countdownInterval) clearInterval(countdownInterval);
      countdownInterval = setInterval(() => {
        secondsLeft -= 1;
        if (secondsLeft <= 0) {
          finishSuccessEarly();
        } else {
          timerEl.textContent = secondsLeft;
        }
      }, 1000);
    }

    }
